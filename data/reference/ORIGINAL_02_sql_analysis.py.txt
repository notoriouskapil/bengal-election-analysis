# ============================================================
# Bengal Election 2021 vs 2026 — SQL Analysis Script
# Step 2 of 3 in the portfolio project pipeline
# ============================================================
# What this script does:
#   1. Loads clean CSVs into a SQLite database
#   2. Runs SQL queries for each analysis angle
#   3. Saves all results to analysis_results.xlsx (one sheet per query)
# ============================================================

import pandas as pd
import sqlite3
import os

# ── Paths ────────────────────────────────────────────────────
BASE    = os.path.dirname(os.path.abspath(__file__))
OUT_DIR = os.path.join(BASE, "bengal", "clean_output")
# SQLite DB lives in /tmp during the run (avoids mounted-filesystem I/O limits)
# A copy is saved to OUT_DIR at the end for your portfolio
DB_PATH = "/tmp/bengal_elections.db"
EXCEL_OUT = os.path.join(BASE, "bengal", "clean_output", "analysis_results.xlsx")

# ── Load CSVs ────────────────────────────────────────────────
clean21   = pd.read_csv(os.path.join(OUT_DIR, "clean_2021.csv"))
clean26   = pd.read_csv(os.path.join(OUT_DIR, "clean_2026.csv"))
winners21 = pd.read_csv(os.path.join(OUT_DIR, "winners_2021.csv"))
winners26 = pd.read_csv(os.path.join(OUT_DIR, "winners_2026.csv"))

# ── Create SQLite DB ─────────────────────────────────────────
conn = sqlite3.connect(DB_PATH)

clean21.to_sql("candidates_2021",  conn, if_exists="replace", index=False)
clean26.to_sql("candidates_2026",  conn, if_exists="replace", index=False)
winners21.to_sql("winners_2021",   conn, if_exists="replace", index=False)
winners26.to_sql("winners_2026",   conn, if_exists="replace", index=False)

print("SQLite DB created at:", DB_PATH)

# ════════════════════════════════════════════════════════════
# SQL QUERIES
# ════════════════════════════════════════════════════════════

queries = {}

# ── Q1: Party-wise seats won — 2021 vs 2026 ─────────────────
queries["Party_Seats_Comparison"] = """
SELECT
    COALESCE(w21.party, w26.party) AS party,
    COALESCE(w21.seats_2021, 0)    AS seats_2021,
    COALESCE(w26.seats_2026, 0)    AS seats_2026,
    COALESCE(w26.seats_2026, 0) - COALESCE(w21.seats_2021, 0) AS seat_change
FROM
    (SELECT party, COUNT(*) AS seats_2021
     FROM winners_2021 GROUP BY party) w21
FULL OUTER JOIN
    (SELECT party, COUNT(*) AS seats_2026
     FROM winners_2026 GROUP BY party) w26
ON w21.party = w26.party
ORDER BY seats_2026 DESC
"""

# SQLite doesn't support FULL OUTER JOIN — simulate with UNION
queries["Party_Seats_Comparison"] = """
SELECT party,
       SUM(seats_2021) AS seats_2021,
       SUM(seats_2026) AS seats_2026,
       SUM(seats_2026) - SUM(seats_2021) AS seat_change
FROM (
    SELECT party, COUNT(*) AS seats_2021, 0 AS seats_2026
    FROM winners_2021 GROUP BY party
    UNION ALL
    SELECT party, 0 AS seats_2021, COUNT(*) AS seats_2026
    FROM winners_2026 GROUP BY party
)
GROUP BY party
ORDER BY seats_2026 DESC
"""

# ── Q2: Party-wise total vote share 2021 vs 2026 ────────────
queries["Party_Vote_Share"] = """
SELECT party,
       SUM(CASE WHEN year = 2021 THEN total_votes ELSE 0 END) AS votes_2021,
       SUM(CASE WHEN year = 2026 THEN total_votes ELSE 0 END) AS votes_2026,
       ROUND(
           100.0 * SUM(CASE WHEN year = 2021 THEN total_votes ELSE 0 END)
           / (SELECT SUM(total_votes) FROM candidates_2021 WHERE party != 'NOTA'), 2
       ) AS vote_pct_2021,
       ROUND(
           100.0 * SUM(CASE WHEN year = 2026 THEN total_votes ELSE 0 END)
           / (SELECT SUM(total_votes) FROM candidates_2026 WHERE party != 'NOTA'), 2
       ) AS vote_pct_2026
FROM (
    SELECT party, total_votes, year FROM candidates_2021 WHERE party != 'NOTA'
    UNION ALL
    SELECT party, total_votes, year FROM candidates_2026 WHERE party != 'NOTA'
)
GROUP BY party
HAVING SUM(total_votes) > 100000
ORDER BY votes_2026 DESC
"""

# ── Q3: Constituencies that FLIPPED between parties ─────────
queries["Flipped_Seats"] = """
SELECT
    w21.ac_no,
    w21.ac_name,
    w21.party    AS winner_2021,
    w26.party    AS winner_2026,
    w21.total_votes AS votes_2021,
    w26.total_votes AS votes_2026,
    w21.win_margin  AS margin_2021,
    w26.win_margin  AS margin_2026
FROM winners_2021 w21
JOIN winners_2026 w26 ON w21.ac_no = w26.ac_no
WHERE w21.party != w26.party
ORDER BY w21.party, w26.party
"""

# ── Q4: Seats held by same party both years ──────────────────
queries["Retained_Seats"] = """
SELECT
    w21.ac_no,
    w21.ac_name,
    w21.party AS party,
    w21.total_votes AS votes_2021,
    w26.total_votes AS votes_2026,
    w26.total_votes - w21.total_votes AS vote_change
FROM winners_2021 w21
JOIN winners_2026 w26 ON w21.ac_no = w26.ac_no
WHERE w21.party = w26.party
ORDER BY w21.party, w21.ac_name
"""

# ── Q5: Top 10 biggest win margins 2026 ─────────────────────
queries["Biggest_Wins_2026"] = """
SELECT ac_name, party, candidate, total_votes,
       runner_up_candidate, runner_up_party, runner_up_votes, win_margin
FROM winners_2026
ORDER BY win_margin DESC
LIMIT 10
"""

# ── Q6: Top 10 closest fights 2026 ──────────────────────────
queries["Closest_Fights_2026"] = """
SELECT ac_name, party, candidate, total_votes,
       runner_up_candidate, runner_up_party, runner_up_votes, win_margin
FROM winners_2026
WHERE win_margin IS NOT NULL
ORDER BY win_margin ASC
LIMIT 10
"""

# ── Q7: Gender comparison of winners ────────────────────────
queries["Gender_Analysis"] = """
SELECT
    gender,
    SUM(CASE WHEN year = 2021 THEN 1 ELSE 0 END) AS winners_2021,
    SUM(CASE WHEN year = 2026 THEN 1 ELSE 0 END) AS winners_2026
FROM (
    SELECT gender, 2021 AS year FROM winners_2021
    UNION ALL
    SELECT gender, 2026 AS year FROM winners_2026
)
GROUP BY gender
"""

# ── Q8: Constituency-level vote swing summary ────────────────
queries["Constituency_Swing"] = """
SELECT
    w21.ac_no,
    w21.ac_name,
    w21.party      AS party_2021,
    w26.party      AS party_2026,
    w21.total_votes AS winner_votes_2021,
    w26.total_votes AS winner_votes_2026,
    w21.win_margin  AS margin_2021,
    w26.win_margin  AS margin_2026,
    CASE WHEN w21.party = w26.party THEN 'Retained' ELSE 'Flipped' END AS result
FROM winners_2021 w21
JOIN winners_2026 w26 ON w21.ac_no = w26.ac_no
ORDER BY w21.ac_no
"""

# ── Q9: Avg win margin by party 2026 ────────────────────────
queries["Avg_Margin_By_Party_2026"] = """
SELECT party,
       COUNT(*) AS seats_won,
       ROUND(AVG(win_margin), 0) AS avg_win_margin,
       MAX(win_margin) AS max_margin,
       MIN(win_margin) AS min_margin
FROM winners_2026
WHERE win_margin IS NOT NULL
GROUP BY party
HAVING COUNT(*) >= 2
ORDER BY seats_won DESC
"""

# ════════════════════════════════════════════════════════════
# RUN QUERIES & SAVE TO EXCEL
# ════════════════════════════════════════════════════════════
print("\nRunning SQL queries and saving to Excel...")

with pd.ExcelWriter(EXCEL_OUT, engine="openpyxl") as writer:
    for sheet_name, sql in queries.items():
        df_result = pd.read_sql_query(sql, conn)
        df_result.to_excel(writer, sheet_name=sheet_name, index=False)
        print(f"  ✓ {sheet_name}: {len(df_result)} rows")

conn.close()

# Copy DB to output folder for portfolio
import shutil
shutil.copy(DB_PATH, os.path.join(OUT_DIR, "bengal_elections.db"))

print(f"\n✓ Analysis complete. Results saved to:\n  {EXCEL_OUT}")
print(f"  SQLite DB: {os.path.join(OUT_DIR, 'bengal_elections.db')}")
